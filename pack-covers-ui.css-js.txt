function packSlugFromName(name) {
            if (!name) return '';
            const map = {
                'Dog': 'dog', 'Lunch': 'lunch', 'Bug': 'bug', 'Pirate': 'pirate',
                'Outback': 'outback', 'Ice Monster': 'iceMonster', 'Dino': 'dino',
                'Aquatic': 'aquatic', 'Safari': 'safari', 'Bot': 'bot', 'Space': 'space',
                'Medieval': 'medieval', 'Wonderland': 'wonderland', 'Breakfast': 'breakfast',
                'Spooky': 'spooky', 'Autumn': 'autumn', 'Blizzard': 'blizzard',
                '?': 'mystery', 'Mystery': 'mystery',
                'Lovely': 'lovely', 'Lucky': 'lucky', 'Spring': 'spring'
            };
            if (map[name]) return map[name];
            return String(name).toLowerCase().replace(/[^a-z0-9]+/g, '');
        }


function packBackgroundUrl(nameOrSlug) {
            const slug = packSlugFromName(nameOrSlug) || String(nameOrSlug || '').replace(/[^a-zA-Z0-9]/g, '');
            if (!slug) return '';
            if (slug === 'mystery') {
                return 'https://static.wikia.nocookie.net/blooket/images/d/d7/Color_Pack_Background.jpg/revision/latest?cb=20240903105616';
            }
            return 'https://ac.blooket.com/blookassets/packs/backgrounds/' + slug + '.svg';
        }


function packCardBackgroundStyle(name) {
            const slug = packSlugFromName(name);
            if (slug === 'lovely') {
                return 'background:radial-gradient(circle at 50% 40%, #ffc4d0 0%, #ff8fab 42%, #f43f7a 100%);';
            }
            if (slug === 'lucky') {
                return 'background:radial-gradient(circle at 50% 40%, #9dffb8 0%, #34d399 42%, #059669 100%);';
            }
            if (slug === 'spring') {
                return 'background:radial-gradient(circle at 50% 40%, #a5e4ff 0%, #38bdf8 42%, #0284c7 100%);';
            }
            if (slug === 'mystery') {
                return "background-image:url('https://static.wikia.nocookie.net/blooket/images/d/d7/Color_Pack_Background.jpg/revision/latest?cb=20240903105616');background-size:cover;background-position:center;background-color:#9ca3af;";
            }
            const url = packBackgroundUrl(name);
            if (!url) return '';
            return "background-image:url('" + url + "');background-size:cover;background-position:center;background-color:transparent;";
        }


function sealedPackUrl(slug) {
            // Full pack art (includes printed top — we clip it in CSS)
            try {
                const hit = (typeof packSimCatalog !== 'undefined') && packSimCatalog.find(function(x){ return x.slug === slug; });
                if (hit && hit.img) return hit.img;
            } catch (e) {}
            return 'https://ac.blooket.com/blookassets/packs/' + slug + '.png';
        }


function renderPackSim() {
            const grid = document.getElementById('packsim-grid');
            if (!grid) return;
            grid.innerHTML = packSimCatalog.map((p, i) => {
                const icon = p.icon || ('https://ac.blooket.com/blookassets/packs/icons/' + p.slug + '.svg');
                const img = p.img || ('https://ac.blooket.com/blookassets/packs/' + p.slug + '.png');
                const isMystery = (p.name === '?' || p.slug === 'mystery');
                let wrapBg = (typeof packCardBackgroundStyle === 'function')
                    ? packCardBackgroundStyle(p.name)
                    : ('background:' + (p.gradient || '') + ';');
                if (!wrapBg) wrapBg = 'background:' + (p.gradient || 'radial-gradient(circle,#0bc2cf,#349aef)') + ';';
                const cornerHtml = isMystery
                    ? '<div class="ps-pack-letter">?</div>'
                    : ('<img alt="' + String(p.name).replace(/"/g, '') + '" class="ps-pack-shadow" src="' + icon + '" width="80" height="80" draggable="false" loading="lazy">');
                return `
                <div class="ps-pack" tabindex="0" role="button" title="${p.name}" onclick="onPackSimClick(${i})" onkeydown="if(event.key==='Enter'||event.key===' '){event.preventDefault();onPackSimClick(${i})}">
                    <div class="ps-pack-img-wrap" style="${wrapBg}">
                        ${cornerHtml}
                        <img src="${img}" alt="${p.name}" class="ps-pack-img" width="150" height="165" draggable="false" loading="lazy">
                    </div>
                    <div class="ps-pack-bottom">
                        <img alt="Token" src="${TOKEN_ICON}" width="23" height="23" draggable="false" loading="lazy">
                        <span class="ps-pack-price">${p.price}</span>
                    </div>
                    <div class="ps-pack-overlay"></div>
                </div>`;
            }).join('');
        }


function renderMarket() {
            const grid = document.getElementById('market-grid');
            if (!grid) return;
            grid.innerHTML = marketPacks.map((p, i) => {

                <div class="pack-card" onclick="openPackModal(${i})" role="button" tabindex="0" title="${p.name}"${styleAttr}>
                    ${corner}
                    <img src="${p.icon}" alt="${p.name}" loading="lazy">
                </div>`;
            }).join('');
        }
